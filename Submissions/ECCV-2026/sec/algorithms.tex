\section{H-SFP Algorithms}
\label{app:algorithm}
Here, we provide a detailed description of the algorithms for each tier of the H-SFP framework. The core logic is a \textbf{dual-timescale hybrid process}:
\begin{enumerate}
    \item A \textbf{fast-timescale} (run every round) feed-forward, generative workflow where tiers exchange lightweight feature statistics ($\mu, \sigma$).
    \item A \textbf{slow-timescale} (run at intervals $I_c$ and $I_e$) background process that performs standard model averaging to prevent long-term model drift.
\end{enumerate}

\subsection*{Algorithm 1: Client-Side Training}
\begin{equation}
    \label{eq:supcon}
    \mathcal{L}_{k}(W_{c,k})= \mathcal{L}_{\text{con}, k} = \sum_{i\in B}\frac{-1}{|P(i)|}\sum_{p\in P(i)}\log\frac{\exp(z_{i}\cdot z_{p}/\tau)}{\sum_{a\in A(i)}\
    exp(z_{i}\cdot z_{a}/\tau)},
\end{equation}
\Cref{alg:client} details the client-side procedure. Each client $k$ receives the current global client-side model $W_c$ from its parent edge server (Line 3). It then trains this model locally for $E$ epochs on its private data $D_k$ (Line 4). The training objective is a self-supervised contrastive loss, $\mathcal{L}_{\text{con}}$ (Line 7), which is formally defined in \Cref{eq:supcon}. This objective learns a robust feature space by pulling features from the same class together and pushing others apart.

After local training is complete, the client "packages" its learned feature space. It computes the class-conditional mean $\mu_k^{(j)}$ (\Cref{mean_client}) and standard deviation $\sigma_k^{(j)}$ (\Cref{std_client}) for all classes $\mathcal{Y}_k$ in its local data (Line 11). Finally, the client returns both its updated local model $W_{c,k}$ (for the slow-timescale averaging) and its computed feature statistics $\text{Stats}_k$ (for the fast-timescale synthesis) to its parent edge server (Line 12).

\begin{algorithm}[!ht]
    \caption{H-SFP: Client-Side Training}
    \label{alg:client}
    \begin{algorithmic}[1]
    \State \textbf{Input:} Client $k$, local data $D_k$, learning rate $\eta$, epochs $E$.
    \State \textbf{Receive:} Client model $W_c$.
    \State Set $W_{c,k} \leftarrow W_c$.
    \For{epoch $e = 1$ to $E$}
        \For{batch $(x, y)$ in $D_k$}
            \State $z \leftarrow W_{c,k}(x)$ \Comment{Compute feature embeddings}
            \State $\mathcal{L}_{\text{con}} \leftarrow \text{ContrastiveLoss}(z, y)$ \Comment{See \Cref{eq:supcon}}
            \State $W_{c,k} \leftarrow W_{c,k} - \eta \nabla \mathcal{L}_{\text{con}}$
        \EndFor
    \EndFor
    \State Compute final local statistics $\text{Stats}_k = \{(\mu_k^{(j)}, \sigma_k^{(j)}) \mid j \in \mathcal{Y}_k\}$
    \Comment{See \Cref{mean_client,std_client}}
    \State \textbf{Return:} $W_{c,k}$, $\text{Stats}_k$ to Edge Server $m$.
    \end{algorithmic}
\end{algorithm}


\begin{equation}
    \mu_k^{(j)} = \frac{1}{|\mathcal{D}_k^{(j)}|} \sum_{x_i \in \mathcal{D}_k^{(j)}} W_{c,k}(x_i),
    \label{mean_client}
\end{equation}
\begin{equation}
    \sigma_k^{(j)} = \sqrt{\frac{1}{|\mathcal{D}_k^{(j)}|} \sum_{x_i \in \mathcal{D}_k^{(j)}} (W_{c,k}(x_i) - \mu_k^{(j)})^2}.
    \label{std_client}
\end{equation}
\begin{equation}
    \label{eq:synthesis}
    z_{syn} \sim \mathcal{N}(\mu_m^{(j)}, \text{diag}(\sigma_m^{(j)} + \epsilon)^2),
\end{equation}
\subsection*{Algorithm 2: Edge Server Logic}

\Cref{alg:edge} describes the logic for the edge server $m$, which manages both the fast-timescale generative process and the slow-timescale model averaging.

\begin{itemize}
    \item \textbf{Fast-Timescale (Statistics Aggregation \& Training):} In every round $t$, the edge server first receives the feature statistics $\text{Stats}_k$ from all clients in its cluster $\mathcal{C}_m$ (Line 5). It aggregates these statistics into a stable, cluster-level distribution $(\mu_m, \sigma_m)$ using the mean aggregation (\Cref{mean_edge}) and variance aggregation (\Cref{std_edge}) (Line 6). Using this distribution, it synthesizes a new training dataset, $D_{\text{syn}}^{\text{edge}}$ (Line 8), by sampling from a multivariate Gaussian with a diagonal covariance, as shown in \Cref{eq:synthesis}. The edge server then trains its own model segment, $W_{e,m}$, using the same contrastive loss objective (\Cref{eq:supcon}) on this synthetic data (Lines 9-11). It then computes the statistics of its \textit{own} output features and forwards these to the cloud (Lines 13-14).

    \item \textbf{Slow-Timescale (Model Averaging):} The edge server orchestrates the periodic model averaging. At an interval of $I_c$ (e.g., every 5 rounds), it receives the local \textit{models} $W_{c,k}$ from its clients, performs FedAvg to create a new global $W_c$, and distributes it back (Lines 16-19). Similarly, at an interval of $I_e$, it sends its own trained model $W_{e,m}$ to the cloud for global averaging (Lines 21-23). It also receives the updated global $W_e$ from the cloud at the start of the round (Line 2).
\end{itemize}

\begin{algorithm}[!ht]
\caption{H-SFP: Edge Server Logic (Server $m$ at round $t$)}
\label{alg:edge}
\begin{algorithmic}[1]
    \State \textbf{Input:} Edge model $W_e$, learning rate $\eta_e$, intervals $I_c, I_e$.
    \State \textbf{Receive from Cloud:} Edge model $W_e$ \Comment{Slow-timescale model update}
    \State $W_{e,m} \leftarrow W_e$
    \
    \State \text{// 1. Fast-Timescale: Receive and Aggregate Statistics}
    \State Receive $\{Stats_k\}_{k \in \mathcal{C}_m}$ from all clients in cluster $\mathcal{C}_m$.
    \State $(\mu_m, \sigma_m) \leftarrow (\{Stats_k\}_{k \in \mathcal{C}_m})$ \Comment{Using \Cref{mean_edge,std_edge}}
    \
    \State \text{// 2. Synthesize and Train}
    \State $D_{syn}^{edge} \leftarrow (\mu_m, \sigma_m)$ \Comment{Using \Cref{eq:synthesis}}
    \State $z'_{syn} \leftarrow W_{e,m}(D_{syn}^{edge})$
    \State $\mathcal{L}_{con,m} \leftarrow \text{ContrastiveLoss}(z'_{syn}, y_{syn})$ \Comment{See \Cref{eq:supcon}}
    \State $W_{e,m} \leftarrow W_{e,m} - \eta_e \nabla \mathcal{L}_{con,m}$
    \
    \State \text{// 3. Package and Send Statistics to Cloud}
    \State $Stats_m \leftarrow (z'_{syn}, y_{syn})$ \Comment{Compute output stats $(\mu_m', \sigma_m')$ }
    \State \textbf{Send to Cloud:} $Stats_m$
    \
    \State \text{// 4. Slow-Timescale: Client Model Averaging}
    \If{$t \mod I_c == 0$}
        \State Receive $\{W_{c,k}\}_{k \in \mathcal{C}_m}$ from clients.
        \State $W_c \leftarrow \frac{1}{|\mathcal{C}_m|} \sum_{k \in \mathcal{C}_m} W_{c,k}$
        \State \textbf{Send to all $k \in \mathcal{C}_m$:} $W_c$
    \EndIf
    \
    \State \text{// 5. Slow-Timescale: Send Edge Model to Cloud}
    \If{$t \mod I_e == 0$}
        \State \textbf{Send to Cloud:} $W_{e,m}$
    \EndIf
\end{algorithmic}
\end{algorithm}

\begin{equation}
    \mu_m^{(j)} = \frac{1}{|\mathcal{C}_m|} \sum_{k \in \mathcal{C}_m} \mu_k^{(j)},
    \label{mean_edge}
\end{equation}
\begin{equation}
    (\sigma_m^{(j)})^2 = \frac{1}{|\mathcal{C}_m|} \sum_{k \in \mathcal{C}_m} (\sigma_k^{(j)})^2.
    \label{std_edge}
\end{equation}


\subsection*{Algorithm 3: Cloud Server Logic}

\Cref{alg:cloud} details the cloud server logic, which mirrors the edge server's generative process but for the final classification task.

\begin{itemize}
    \item \textbf{Fast-Timescale (Final Classification):} In every round, the cloud receives the aggregated output statistics $\text{Stats}_m$ from all $M$ edge servers (Line 3). It aggregates these into a final global distribution $(\mu_g, \sigma_g)$ (Line 4) (using an aggregation method analogous to \Cref{mean_edge,std_edge}) and synthesizes a global dataset $D_{\text{syn}}^{\text{global}}$ (Line 6) (analogous to \Cref{eq:synthesis}). Critically, the cloud trains the final classifier $W_g$ not with a contrastive loss, but with a standard supervised \textbf{Cross-Entropy Loss ($\mathcal{L}_{CE}$)} on this synthetic data (Lines 7-9). This trains the model for the end-task.

    \item \textbf{Slow-Timescale (Model Averaging):} At the $I_e$ interval, the cloud receives the edge models $\{W_{e,m}\}$, averages them into a new global $W_e$, and distributes this model back to all edge servers (Lines 11-14).
\end{itemize}

\begin{algorithm}[!ht]
\caption{H-SFP: Cloud Server Logic (at round $t$)}
\label{alg:cloud}
\begin{algorithmic}[1]
    \State \textbf{Input:} Global classifier $W_g$, learning rate $\eta_g$, interval $I_e$.
    \
    \State \text{// 1. Fast-Timescale: Receive and Aggregate Statistics}
    \State Receive $\{Stats_m\}_{m=1}^M$ from all $M$ edge servers.
    \State $(\mu_g, \sigma_g) \leftarrow (\{Stats_m\}_{m=1}^M)$ \Comment{Analogous to \Cref{mean_edge,std_edge}}
    \
    \State \text{// 2. Synthesize and Train Final Classifier}
    \State $D_{syn}^{global} \leftarrow (\mu_g, \sigma_g)$ \Comment{Analogous to \Cref{eq:synthesis}}
    \State $logits \leftarrow W_g(D_{syn}^{global})$
    \State $\mathcal{L}_{CE} \leftarrow CrossEntropyLoss(logits, y_{syn})$ \Comment{Standard CE Loss}
    \State $W_g \leftarrow W_g - \eta_g \nabla \mathcal{L}_{CE}$
    \
    \State \text{// 3. Slow-Timescale: Edge Model Averaging}
    \If{$t \mod I_e == 0$}
        \State Receive $\{W_{e,m}\}_{m=1}^M$ from edge servers.
        \State $W_e \leftarrow \frac{1}{M} \sum_{m=1}^M W_{e,m}$
        \State \textbf{Send to all $m$:} $W_e$
    \EndIf
\end{algorithmic}
\end{algorithm}
